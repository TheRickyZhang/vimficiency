// n -> m, 10 -> 1, i+1 -> 2 * i

#include <bits/stdc++.h>
using namespace std;

int main() {
  int end = 0;
  int n; cin >> n;
  vector<int> a(n);
  for(int i = 0; i < n; i++) {
    cin >> a[i];
    if(i < n/2) {
      a[i * 2] += a[i];
    }
  }
  cout << max_element(a.begin(), a.end()) << endl;
}
